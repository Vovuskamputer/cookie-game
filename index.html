📱 Как играть в Telegram:

1. Создай бота через @BotFather

2. Используй этот код для бота:

// Telegram бот для Cookie Empire
const TelegramBot = require('node-telegram-bot-api');
const TOKEN = 'ТВОЙ_ТОКЕН';

const bot = new TelegramBot(TOKEN, {polling: true});

bot.onText(/\/start/, (msg) => {
    const chatId = msg.chat.id;
    bot.sendMessage(chatId, '🎮 Cookie Empire - крутая игра! 🍪', {
        reply_markup: {
            inline_keyboard: [[{
                text: '🎮 ИГРАТЬ',
                url: 'https://ТВОЙНИК.github.io/cookie-empire/game.html'
            }]]
        }
    });
});
