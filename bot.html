🤖 Код Telegram бота для Cookie Empire

Скопируй этот код и используй на Glitch.com или другом хостинге:

📝 Файл: bot.js

const TelegramBot = require('node-telegram-bot-api');

// ⚠️ ЗАМЕНИ ЭТО НА СВОЙ ТОКЕН ОТ @BotFather!
const TOKEN = '7974037858:AAF8nR1MGuLRA9Dz_MLPYZmTnWjtNpccRyk';

// ⚠️ ЗАМЕНИ ЭТО НА СВОЙ URL С GITHUB PAGES!
const GAME_URL = 'https://vovuskamputer.github.io/cookie-game/';

const bot = new TelegramBot(TOKEN, { polling: true });
console.log('✅ Бот запущен!');

// Команда /start
bot.onText(/\/start/, (msg) => {
    const chatId = msg.chat.id;
    const name = msg.from.first_name || 'Игрок';
    
    bot.sendMessage(chatId, 
`🎮 *Привет, ${name}!* 🍪

*Cookie Empire* - самая крутая игра про печеньки!

*Твоя ссылка на игру:*
${GAME_URL}

Нажми кнопку ниже чтобы начать!`, 
    {
        parse_mode: 'Markdown',
        reply_markup: {
            inline_keyboard: [[{
                text: '🎮 ИГРАТЬ В COOKIE EMPIRE',
                url: GAME_URL
            }]]
        }
    });
});

// Команда /play
bot.onText(/\/play/, (msg) => {
    const chatId = msg.chat.id;
    const userId = msg.from.id;
    const name = msg.from.first_name || 'Игрок';
    
    // Ссылка с параметрами пользователя
    const gameLink = `${GAME_URL}?player=tg_${userId}&name=${encodeURIComponent(name)}`;
    
    bot.sendMessage(chatId, '🎮 Открываю игру...', {
        reply_markup: {
            inline_keyboard: [[{
                text: '🎮 ОТКРЫТЬ ИГРУ',
                url: gameLink
            }]]
        }
    });
});
